    Agonet 使用指南¶

    面向业务开发者：快速开始、编写 Handler、配置、最佳实践

    快速开始¶

    方式一：核心层 EventHandler（简单场景）¶

    import (
    "github/aif-go/ag-core/contribute/agonet"
)

// 1. 实现 EventHandler
type EchoHandler struct {
    agonet.BuiltinEventEngine  // 内嵌空实现
}

func (h *EchoHandler) OnTraffic(c agonet.Conn) (action agonet.Action) {
    // 读取所有可用数据
    buf, err := c.Next(c.InboundBuffered())
    if err != nil {
        return agonet.Close
    }
    // 写回数据
    _, _ = c.Write(buf)
    return agonet.None
}

// 2. 启动服务器
func main() {
    handler := &EchoHandler{}
    config := agonet.DefaultServerConfig()
    config.Addr = "tcp://:9000"

    server, _ := agonet.NewServer(handler, &config)
    server.Start()  // 阻塞
}

    方式二：Simple 管道层（推荐）¶

    import (
    "github/aif-go/ag-core/contribute/agonet"
    "github/aif-go/ag-core/contribute/agonet/simple"
)

// 1. 定义 Handler（嵌入 SimpleHandler 避免实现所有接口）
type EchoHandler struct{}

func (h *EchoHandler) HandleRead(ctx simple.InboundContext, msg any) {
    reader := msg.(agonet.Reader)
    data, _ := reader.Next(reader.InboundBuffered())
    ctx.Write(data) // 通过 Outbound 管道写回 → head → TCP
}

// 2. 创建 SimpleEventHandler
handler, _ := simple.NewSimpleEventHandlerWithOptions(
    simple.WithChannelInitializer(func(channel simple.Channel) error {
        channel.Pipeline().AddLast(&EchoHandler{})
        return nil
    }),
)

// 3. 启动
config := agonet.DefaultServerConfig()
config.Addr = "tcp://:9000"
server, _ := agonet.NewServer(handler, &config)
server.Start()

    配置示例¶

    最小 TCP 服务端¶

    addr: "tcp://:9000"
config:
  engine:
    multicore: true

    TLS 服务端¶

    addr: "tcp://:9443"
config:
  security:
    type: tls
    certsDir: /etc/certs
    tls:
      caPath: ca.crt
      authCertPath: server.crt
      authKeyPath: server.key

    TLCP（国密）服务端¶

    addr: "tcp://:9443"
config:
  security:
    type: tlcp
    certsDir: /etc/certs
    tlcp:
      caPath: sm2-ca.cer
      authCertPath: sm2-auth.cer
      authKeyPath: sm2-auth.key
      signCertPath: sm2-sign.cer
      signKeyPath: sm2-sign.key
      encCertPath: sm2-enc.cer
      encKeyPath: sm2-enc.key

    双栈服务端（同时支持 TLS + TLCP）¶

    addr: "tcp://:9443"
config:
  security:
    type: tls_tlcp
    certsDir: /etc/certs
    tls:
      caPath: ca.crt
      authCertPath: server.crt
      authKeyPath: server.key
    tlcp:
      caPath: sm2-ca.cer
      authCertPath: sm2-auth.cer
      authKeyPath: sm2-auth.key
      signCertPath: sm2-sign.cer
      signKeyPath: sm2-sign.key
      encCertPath: sm2-enc.cer
      encKeyPath: sm2-enc.key

    Simple 管道模式示例¶

    LengthField 编解码（自定义协议）¶

    import (
    "encoding/binary"
    "github/aif-go/ag-core/contribute/agonet"
    "github/aif-go/ag-core/contribute/agonet/simple"
)

handler, _ := simple.NewSimpleEventHandlerWithOptions(
    simple.WithChannelInitializer(func(channel simple.Channel) error {
        channel.Pipeline().AddLast(
            // 入站：LengthField 拆包（前 4 字节为长度字段）
            simple.NewLengthFieldDecoder(
                binary.BigEndian, // byteOrder
                1024 * 1024,      // maxFrameLength
                0,                // lengthFieldOffset
                4,                // lengthFieldLength
                0,                // lengthAdjustment
                4,                // initialBytesToStrip
            ),
            // 解码后 msg 已经是完整帧的 []byte
            &BizHandler{},
        )
        return nil
    }),
)

// BizHandler 收到解码后的完整帧
type BizHandler struct{}

func (h *BizHandler) HandleRead(ctx simple.InboundContext, msg any) {
    frame := msg.([]byte) // LengthFieldDecoder 解码后传出 []byte
    // 处理 frame ...
    ctx.Write(response)
}

    泛型 Handler（SimpleInboundHandler[T]）¶

    import "github/aif-go/ag-core/contribute/agonet/simple"

// 自动做类型断言，类型不匹配时透传
handler := simple.NewSimpleInboundHandler(func(ctx simple.InboundContext, frame []byte) {
    // frame 已经是 []byte，无需手动断言
    ctx.Write(process(frame))
})

    FX 集成¶

    import (
    "github/aif-go/ag-core/fxs"
    "github/aif-go/ag-core/contribute/agonet"
    "github/aif-go/ag-core/contribute/agonet/simple"
    "go.uber.org/fx"
)

func main() {
    app := fxs.FxAppMode(
        fxs.FxAgConfModule,

        // Server: 注入 EventHandler → FxAgonetServerModule 自动装配
        fx.Provide(
            func() agonet.EventHandler {
                handler, _ := simple.NewSimpleEventHandlerWithOptions(
                    simple.WithChannelInitializer(func(ch simple.Channel) error {
                        ch.Pipeline().AddLast(&MyBizHandler{})
                        return nil
                    }),
                )
                return handler
            },
        ),
        agonet.FxAgonetServerModule,

        // Client
        agonet.FxAgonetClientModule,

        // ShortClient: 用 group tag 注入 Option
        fx.Provide(
            simple.FxAgonetSimpleGroupTag(func() simple.ShortClientOption {
                return simple.WithShortClientTimeout(5 * time.Second)
            }),
        ),
        fx.Provide(simple.NewShortClientFx),
    )
    app.Run()
}

    FxAgonetServerModule 提供 ServerConfig + Server + WarpServer。 FxAgonetClientModule 提供 ClientConfig + Client。 两者都依赖 ag_conf.IEnvironment（需先加载 FxAgConfModule）。

    最佳实践¶

    ✅ Simple 管道层优先¶

    除极简单场景外，优先使用 simple 管道层。它将事件驱动转换为 Handler 链，便于分离编解码和业务逻辑。

    ✅ 嵌入 chain-friendly 设计¶

    Handler 处理完应调用 ctx.FireRead(msg) 或 ctx.FireWrite(msg) 继续传播。拦截则跳过调用。

    // ✅ 解码后传播到下一个 InboundHandler
func (d *MyDecoder) HandleRead(ctx simple.InboundContext, msg any) {
    frame := decode(msg)
    ctx.FireRead(frame) // → 下一个 InboundHandler
}

// ✅ 不需要传播的场景（如 Echo）
func (h *EchoHandler) HandleRead(ctx simple.InboundContext, msg any) {
    data := readData(msg)
    ctx.Write(data) // 写回，不再传播
}

    ✅ 按需求选择 EventLoop 数量¶

    engine:
  multicore: true      # = NumCPU，大部分场景适用
  numEventLoop: 4      # 手动指定（如 IO 密集型业务）

    ✅ 合理配置 KeepAlive¶

    keepAlive:
  enable: true
  idle: 60        # 60 秒无数据开始探测
  interval: 12    # 每 12 秒探测一次
  count: 5        # 5 次失败断开（共 60s 探测周期）

    ❌ Handler 中不要做阻塞操作¶

    EventHandler 和 Handler 都在 EventLoop 的 goroutine 中执行，阻塞操作会阻塞整个 EventLoop。

    // ❌ 不要这样做
func (h *Handler) HandleRead(ctx simple.InboundContext, msg any) {
    result := heavyDatabaseQuery(msg)  // 阻塞 EventLoop！
    ctx.Write(result)
}

// ✅ 使用 EventLoop.Execute() 异步执行
func (h *Handler) HandleRead(ctx simple.InboundContext, msg any) {
    ctx.Channel().EventLoop().Execute(context.Background(),
        agonet.RunnableFunc(func(ctx context.Context) error {
            result := heavyDatabaseQuery(msg)
            // 异步结果通过 conn.Wake() 触发 OnTraffic 处理
            return nil
        }),
    )
}

    ❌ 不要直接读写 Conn 的 net.Conn¶

    // ❌ 直接读写底层连接会导致双缓冲状态不一致
c.NetConn().Read(buf)

// ✅ 通过 Conn 接口提供的方法
c.Next(n)
c.Write(data)

    ❌ 不要在 InboundHandler 中返回 error¶

    当前 Handler 方法无返回值（HandleRead、HandleWrite 等均无 error 返回）。异常应通过 ctx.Channel().Pipeline().FireChannelException(err) 传播到 ExceptionHandler 链。

    // ❌ func (h *Handler) HandleRead(...) error { ... }
// ✅ func (h *Handler) HandleRead(ctx simple.InboundContext, msg any) { ... }

    FAQ¶

    Q: 支持哪些传输协议？ A: tcp、tcp4、tcp6、unix。UDP 代码在 listener 中已注释，不可用。

    Q: TLS 和 TLCP 如何选择？ A: 标准安全通信用 tls；遵循国家密码标准的场景用 tlcp。双栈模式 tls_tlcp 可同时监听两种协议。

    Q: InboundHandler 如何传播消息？ A: 调用 ctx.FireRead(message) 将消息传给下一个 InboundHandler。不调用则拦截。

    Q: ShortClient 和长连接的适用场景？ A: SimpleShortClient 每次请求创建新连接（Promise 模式），适合请求频率低、需要简单 API 的场景。高频通信建议保持长连接并通过 ChannelFromConn 操作 Pipeline。

    Q: OnOpen 返回的 out []byte 在 Simple 层如何工作？ A: SimpleEventHandler 接管了 OnOpen，返回 nil。初始数据应在 Handler 的 HandleActive 中通过 ctx.Write() 发送。直接使用 agonet.EventHandler 层时，OnOpen 的 out 会通过 conn.rawConn.Write(out) 发送，不会经过 Pipeline 和 Codec — 接收端如果使用 LengthField 解码器会收到裸字节导致帧解析失败。
