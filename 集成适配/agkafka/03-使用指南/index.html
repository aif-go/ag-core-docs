    AgKafka 使用指南¶

    面向业务开发者：快速上手、配置、生产者/消费者开发、最佳实践

    快速开始¶

    1. 添加依赖¶

    require (
    github.com/aif-go/ag-core latest
    github.com/IBM/sarama latest
)

    2. 注册 FX 模块¶

    import (
    "github.com/aif-go/ag-core/fxs"
    "github.com/aif-go/ag-core/contribute/agsarama"
)

func main() {
    app := fxs.FxAppMode(
        fxs.FxAgConfModule,
        agsarama.FxAgsaramaModule,
    )
    app.Run()
}

    3. 注入客户端¶

    type MyService struct {
    kafka sarama.Client
}

func NewMyService(result agsarama.FxResult) *MyService {
    // 直接获得可用的 sarama.Client
    return &MyService{kafka: result.Client}
}

    生产者开发¶

    type OrderService struct {
    kafka sarama.Client
}

// 注入客户端
func NewOrderService(result agsarama.FxResult) *OrderService {
    return &OrderService{kafka: result.Client}
}

// 发送消息
func (s *OrderService) PublishOrder(ctx context.Context, order *Order) error {
    producer, err := sarama.NewAsyncProducerFromClient(s.kafka)
    if err != nil {
        return err
    }
    defer producer.Close()

    msg := &sarama.ProducerMessage{
        Topic: "orders",
        Key:   sarama.StringEncoder(order.ID),
        Value: sarama.ByteEncoder(orderJSON),
    }

    producer.Input() <- msg

    select {
    case success := <-producer.Successes():
        slog.Info("produced", "partition", success.Partition, "offset", success.Offset)
        return nil
    case err := <-producer.Errors():
        return err.Err
    case <-ctx.Done():
        return ctx.Err()
    }
}

    同步生产者¶

    ⚠️ 使用 SyncProducer 前，必须在 app.yml 中设置 producer.return.successes: true（默认 false）。否则 NewSyncProducerFromClient 运行时报错。

    func (s *OrderService) PublishSync(ctx context.Context, topic string, value []byte) error {
    producer, err := sarama.NewSyncProducerFromClient(s.kafka)
    if err != nil {
        return err
    }
    defer producer.Close()

    partition, offset, err := producer.SendMessage(&sarama.ProducerMessage{
        Topic: topic,
        Value: sarama.ByteEncoder(value),
    })
    slog.Info("produced", "partition", partition, "offset", offset)
    return err
}

    消费者开发¶

    基础用法（单 Handler）¶

    type OrderConsumer struct {
    kafka sarama.Client
}

func NewOrderConsumer(result agsarama.FxResult) *OrderConsumer {
    return &OrderConsumer{kafka: result.Client}
}

func (c *OrderConsumer) Start(ctx context.Context) error {
    group, err := sarama.NewConsumerGroupFromClient("order-group", c.kafka)
    if err != nil {
        return err
    }
    defer group.Close()

    handler := &OrderHandler{}

    for {
        err := group.Consume(ctx, []string{"orders"}, handler)
        if err != nil {
            return err
        }
        if ctx.Err() != nil {
            return ctx.Err()
        }
    }
}

// 消费者 Handler
type OrderHandler struct{}

func (h *OrderHandler) Setup(session sarama.ConsumerGroupSession) error {
    slog.Info("consumer setup", "memberID", session.MemberID())
    return nil
}

func (h *OrderHandler) Cleanup(session sarama.ConsumerGroupSession) error {
    return nil
}

func (h *OrderHandler) ConsumeClaim(session sarama.ConsumerGroupSession, claim sarama.ConsumerGroupClaim) error {
    for msg := range claim.Messages() {
        slog.Info("consumed",
            "topic", msg.Topic,
            "partition", msg.Partition,
            "offset", msg.Offset,
            "value", string(msg.Value),
        )
        session.MarkMessage(msg, "")  // 确认消费
    }
    return nil
}

    生产环境推荐使用下方的 Consumer 生命周期集成 模式，实现 ag_server.Server 接口 + Multi-Handler 路由。

    手动 ACK（关闭自动提交）¶

    当需要精确控制消费确认时机（如处理成功才 ack、失败不 ack）：

    配置：

    agsarama:
  consumer:
    offsets:
      autoCommit:
        enable: false         # ← 关闭自动提交

    代码：

    func (h *OrderHandler) ConsumeClaim(session sarama.ConsumerGroupSession, claim sarama.ConsumerGroupClaim) error {
    for msg := range claim.Messages() {
        if err := process(msg); err != nil {
            slog.Error("process failed", "offset", msg.Offset, "err", err)
            continue              // 处理失败，不 ack，下次重新消费
        }
        session.MarkMessage(msg, "")  // 处理成功才确认
    }
    return nil
}

    开启自动提交时 MarkMessage 仍有效（标记后等待 Interval 到期提交），关闭后则完全由 MarkMessage + 显式 Commit() 控制。

    Consumer 生命周期集成¶

    Consumer 实现 ag_server.Server 接口，通过 wrapper 归入 group:"ag_servers"，与 Kitex/Hertz 统一启动/停止。

    核心设计：自行管理 context（不依赖 App 传入），Stop 时先 cancel 再 close group，构造阶段完成所有校验。

    配置¶

    routes 定义业务标识→实际 topic 的映射，实现代码与 topic 名的解耦。topics 中的值可以是 route key 或直接 topic 名：

    kafka:
  consumer:
    routes:                               # 业务标识 → 实际 topic
      student-event: student-created      # 生产
      # student-event: dev-student         # 开发环境使用不同 topic
    groups:
      - groupID: "order-group"
        topics: ["student-event", "order-paid"]  # route key / 直接 topic

    Consumer Server¶

    type KafkaConsumerServer struct {
    client  sarama.Client
    handler sarama.ConsumerGroupHandler
    groups  map[string]sarama.ConsumerGroup

    ctx    context.Context
    cancel context.CancelFunc
    config *ConsumerConfig
}

type ConsumerConfig struct {
    Routes map[string]string          // 业务标识 → 实际 topic
    Groups []ConsumerGroupConfig
}
type ConsumerGroupConfig struct {
    GroupID string
    Topics  []string                   // route key 或直接 topic 名
}

// 辅助：解析 route key 为实际 topic
func resolveTopics(topics []string, routes map[string]string) []string {
    var result []string
    for _, t := range topics {
        if actual, ok := routes[t]; ok {
            result = append(result, actual)
        } else {
            result = append(result, t)
        }
    }
    return result
}

func NewKafkaConsumerServer(client sarama.Client, handler *CompositeHandler, config *ConsumerConfig) (*KafkaConsumerServer, error) {
    // 构造阶段集中校验，失败阻止启动
    if len(config.Groups) == 0 {
        return nil, fmt.Errorf("kafka consumer: no groups configured")
    }
    for _, cfg := range config.Groups {
        if len(cfg.Topics) == 0 {
            return nil, fmt.Errorf("kafka consumer: group %q has empty topics", cfg.GroupID)
        }
        actual := resolveTopics(cfg.Topics, config.Routes)
        if len(actual) == 0 {
            return nil, fmt.Errorf("kafka consumer: group %q has no valid topics after route resolution", cfg.GroupID)
        }
        for _, key := range cfg.Topics {
            if !handler.HasHandler(key) {
                return nil, fmt.Errorf("kafka consumer: key %q declared in group %q but no handler registered", key, cfg.GroupID)
            }
        }
    }

    ctx, cancel := context.WithCancel(context.Background())
    return &KafkaConsumerServer{
        client:  client,
        handler: handler,
        groups:  make(map[string]sarama.ConsumerGroup),
        ctx:     ctx,
        cancel:  cancel,
        config:  config,
    }, nil
}

func (s *KafkaConsumerServer) Start(_ context.Context) error {
    for _, cfg := range s.config.Groups {
        actualTopics := resolveTopics(cfg.Topics, s.config.Routes)
        group, err := sarama.NewConsumerGroupFromClient(cfg.GroupID, s.client)
        if err != nil {
            return fmt.Errorf("kafka consumer: create group %q: %w", cfg.GroupID, err)
        }
        s.groups[cfg.GroupID] = group
        go s.consume(cfg.GroupID, group, actualTopics)
    }
    return nil
}

func (s *KafkaConsumerServer) consume(groupID string, group sarama.ConsumerGroup, topics []string) {
    defer func() {
        if err := group.Close(); err != nil {
            slog.Error("consumer group close failed", "group", groupID, "err", err)
        }
    }()
    for {
        select {
        case <-s.ctx.Done():
            return
        default:
            err := group.Consume(context.Background(), topics, s.handler)
            if err != nil {
                if err == sarama.ErrClosedConsumerGroup {
                    return
                }
                slog.Error("consumer error, retry", "group", groupID, "err", err)
                time.Sleep(time.Second)
            }
        }
    }
}

func (s *KafkaConsumerServer) Stop(_ context.Context) error {
    s.cancel()
    for _, group := range s.groups {
        group.Close()
    }
    return nil
}

    Multi-Handler 路由¶

    TopicHandler 接口¶

    Handler 通过 TopicKey() 自声明业务标识，实现代码与 topic 名解耦：

    // internal/kafka/composite.go
type TopicHandler interface {
    TopicKey() string       // 业务标识，如 "student-event"
    Handle(ctx context.Context, msg *sarama.ConsumerMessage) error
}

    CompositeHandler¶

    通过 fx.In 自动收集所有 handler（无需手动 registerHandlers）。reverseRoutes 用于运行时将实际 topic 反向解析为业务标识：

    type CompositeHandlerParams struct {
    fx.In
    Handlers []TopicHandler `group:"kafka_handlers"`
    Config   *ConsumerConfig
}

type CompositeHandler struct {
    handlers       map[string]TopicHandler  // 业务标识 → handler
    reverseRoutes  map[string]string          // 实际 topic → 业务标识
    expectedTopics []string                   // 所有声明的标识
}

func NewCompositeHandler(p CompositeHandlerParams) *CompositeHandler {
    var expected []string
    for _, g := range p.Config.Groups {
        expected = append(expected, g.Topics...)
    }
    reverse := make(map[string]string)
    for k, v := range p.Config.Routes {
        reverse[v] = k
    }

    h := &CompositeHandler{
        handlers:       make(map[string]TopicHandler),
        reverseRoutes:  reverse,
        expectedTopics: expected,
    }
    for _, handler := range p.Handlers {
        key := handler.TopicKey()
        h.handlers[key] = handler
    }
    return h
}

func (h *CompositeHandler) HasHandler(key string) bool {
    _, ok := h.handlers[key]
    return ok
}

func (h *CompositeHandler) Setup(_ sarama.ConsumerGroupSession) error { return nil }
func (h *CompositeHandler) Cleanup(sarama.ConsumerGroupSession) error { return nil }

func (h *CompositeHandler) ConsumeClaim(sess sarama.ConsumerGroupSession, claim sarama.ConsumerGroupClaim) error {
    key := claim.Topic()
    if k := h.reverseRoutes[key]; k != "" {
        key = k                                    // 反向解析：实际 topic → 业务标识
    }
    handler, ok := h.handlers[key]
    if !ok {
        for range claim.Messages() {}              // drain 未知消息
        return fmt.Errorf("no handler for topic: %s", claim.Topic())
    }
    for msg := range claim.Messages() {
        if err := handler.Handle(sess.Context(), msg); err != nil {
            slog.Error("handle failed", "topic", msg.Topic, "err", err)
        }
        sess.MarkMessage(msg, "")                  // 无论成败都 MarkMessage（连续水位语义）
    }
    return nil
}

    FX 模块¶

    Handler 通过 asKafkaHandler 辅助函数自动注入 CompositeHandler：

    // internal/kafka/zfx_kafka.go
func asKafkaHandler(fn any) any {
    return fx.Annotate(fn,
        fx.As(new(TopicHandler)),
        fx.ResultTags(`group:"kafka_handlers"`),
    )
}

var FxKafkaModule = fx.Module("fx-kafka-module",
    fx.Provide(
        NewConsumerConfig,          // *ConsumerConfig — 从 app.yml 绑定
        NewCompositeHandler,        // fx.In: []TopicHandler + *ConsumerConfig
        asKafkaHandler(NewStudentHandler),
        asKafkaHandler(NewOrderHandler),
        NewKafkaConsumerServer,
        fx.Annotate(
            kafkaServerWrapper,
            fx.ResultTags(`group:"ag_servers"`),
        ),
    ),
)

    校验体系¶

    NewKafkaConsumerServer 构造阶段检测以下异常，阻止启动：

    # 校验项 严重度
    ① groups 为空 error（阻止启动）
    ② 任一 group 的 topics 为空 error
    ③ resolveTopics() 结果为空 error
    ④ declared key 无对应 handler error
    ⑤ 多 key 映射同一 topic warn
    ⑥ routes 残留 key（无 group 引用） warn
    ⑦ handler 注册了但无 group 声明 warn

    ConsumeClaim 收到未知 topic 时 drain + 返回 error（阻塞进度，促使修复）。

    最佳实践¶

    ✅ 通过 FX 模块注入 Client¶

    总是通过 agsarama.FxAgsaramaModule 创建和管理 sarama.Client，它自动处理配置绑定和生命周期。

    ✅ 使用 AsyncProducer 应对高吞吐¶

    同步生产者（SendMessage）每次调用都会等待确认，吞吐受限。高吞吐场景使用 AsyncProducer + Successes()/Errors() 通道。

    ✅ 合理配置 Consumer 超时¶

    consumer:
  group:
    session:
      timeout: 10000       # 会话超时，建议 10s
    heartbeat:
      interval: 3000        # 心跳间隔，建议 session.timeout / 3
    rebalance:
      timeout: 60000        # 重平衡超时

    ✅ 幂等生产者防重复¶

    producer:
  idempotent: true          # 启用幂等，防止网络重试导致的消息重复

    ❌ 不要在 consumer 中做耗时操作¶

    // ❌ 这样会导致 rebalance 超时
func (h *Handler) ConsumeClaim(session, claim) error {
    for msg := range claim.Messages() {
        processSlowly(msg)    // 耗时操作
        session.MarkMessage(msg, "")
    }
}

// ✅ 异步处理
func (h *Handler) ConsumeClaim(session, claim) error {
    for msg := range claim.Messages() {
        go processAsync(msg)  // 异步处理
        session.MarkMessage(msg, "")
    }
}

    ❌ 不要忘记 MarkMessage¶

    不调用 MarkMessage 会导致消费者重启时重新消费旧消息。如果需要精确控制 ack 时机（如处理失败不回退），关闭自动提交配合手动 MarkMessage：

    agsarama:
  consumer:
    offsets:
      autoCommit:
        enable: false

    ✅ 复用 sarama.Client，每次创建 producer¶

    通过 agsarama.FxAgsaramaModule 获取全局单例 sarama.Client，每次发消息时从 client 创建 producer 后及时 Close()。不要每次创建新的 sarama.Client（连接开销极大）。

    ✅ Handler 用业务标识注册，topic 名在 YAML 管理¶

    func (h *StudentHandler) TopicKey() string { return "student-event" }

    kafka:
  consumer:
    routes:
      student-event: student-created      # 生产
      # student-event: dev-student         # 开发

    环境切换只改 routes 映射，代码不变。

    ✅ 关闭消费者前先 cancel context，再 close group¶

    Stop() 中先 s.cancel() 停止 consume loop，再 group.Close()，避免死锁和重复 rebalance。

    ❌ 不要在 handler 中硬编码 topic 名¶

    处理失败记录到 DB 或死信队列，不在 handler 中写死 topic 字符串。

    FAQ¶

    Q: 包名为什么是 agsarama 而不是 agkafka？ A: Sarama 是 Go 语言最主流的 Kafka 客户端库，包名直接反映底层依赖。

    Q: 如何配置 SASL 认证？ A: 

    agsarama:
  net:
    sasl:
      enable: true
      mechanism: plain
      user: myuser
      password: mypass

    Q: 如何查看当前 Kafka 版本对应的 Sarama Version？ A: 使用 sarama.ParseKafkaVersion("2.8.0") 字符串格式。不配置时使用 Sarama 默认版本。

    Q: 如何配置从最早偏移量开始消费？ A: 设置 consumer.offsets.initial: -2（-1 = OffsetNewest，-2 = OffsetOldest）。

    Q: 如何关闭自动提交，改为手动 ACK？ A: 

    agsarama:
  consumer:
    offsets:
      autoCommit:
        enable: false
    然后在 Handler 中按需调用 session.MarkMessage(msg, "")，处理失败时不 ack 即可实现重试。

    Q: 配置不生效怎么办？

    A: agsarama 使用 Go 字段名做 EqualFold 大小写不敏感匹配。如果 YAML 中某个配置不生效，对比 key 的拼写与 config.go 中 struct 字段名是否一致。例如 producer.return.successes 绑定时，key successes 必须与 Go 字段 Successes 拼写匹配（大小写可不同，如 SUCCESSES 也可）。
