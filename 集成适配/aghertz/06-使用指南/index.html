    AgHertz 使用指南¶

    面向业务开发者：快速上手、配置、服务端/客户端开发、最佳实践

    快速开始¶

    1. 添加依赖¶

    require (
    github.com/aif-go/ag-core latest
    github.com/cloudwego/hertz latest
    github.com/hertz-contrib/registry/nacos  // 如需 Nacos
)

    2. 注册 FX 模块¶

    import (
    "github.com/aif-go/ag-core/fxs"
    ahserver "github.com/aif-go/ag-core/contribute/aghertz/server"
    ahclient "github.com/aif-go/ag-core/contribute/aghertz/client"
)

func main() {
    app := fxs.FxAppMode(
        // 基础模块
        fxs.FxAgConfModule,
        fxs.FxAgSlogModule,

        // Hertz Server
        ahserver.FxAgHertzServerModule,

        // 注册路由
        fx.Provide(
            ahserver.NewFxServerRouteProvider(func() *ahserver.Route {
                return ahserver.NewRoute("GET", "/api/v1/hello",
                    func(ctx context.Context, c *app.RequestContext) {
                        c.JSON(200, map[string]string{"msg": "hello"})
                    },
                )
            }),
        ),
    )
    app.Run()
}

    配置¶

    Server 配置¶

    # app.yml
hertz:
  server:
    host: 0.0.0.0                 # 绑定 Host
    port: 7000                     # 绑定端口
    adaptivePort: false            # 自适应端口（冲突时自动换端口）
    enableIPRange: ""              # IP 范围过滤（注册用）

    keepAlive: true                # Keep-Alive
    keepAliveTimeout: 60           # Keep-Alive 超时（秒）

    pprof: false                   # 启用 pprof
    pprofPath: /debug/pprof        # pprof 路径

    enableH2C: false               # 启用 H2C 协议

    serviceName: my-service        # 注册的服务名
    # tags:                          # 自定义注册标签
    #   version: v1.0

    # registry:                      # Nacos 注册（可选）
    #   cluster: DEFAULT
    #   group: DEFAULT_GROUP

    Client 配置¶

    # app.yml
hertz:
  client:
    keepAlive: false               # 是否启用 KeepAlive
    dialTimeout: 1000              # 连接超时（ms）
    maxConnsPerHost: 512           # 每主机最大连接数
    maxIdleConnDuration: 10000     # 最大空闲连接时间（ms）

    # discovery:                     # Nacos 服务发现（可选）
    #   enabled: true
    #   type: nacos
    #   nacos:
    #     cluster: DEFAULT
    #     group: DEFAULT_GROUP

    服务端开发¶

    注册路由¶

    // 方式一：在 FX 启动时注册
fx.Provide(
    ahserver.NewFxServerRouteProvider(func() *ahserver.Route {
        return ahserver.NewRoute("GET", "/api/v1/users/:id", handler)
    }),
)

// 方式二：批量注册多个路由
fx.Provide(
    ahserver.NewFxServerRouteProvider(func() []*ahserver.Route {
        return []*ahserver.Route{
            ahserver.NewRoute("GET", "/api/v1/users", ListUsers),
            ahserver.NewRoute("POST", "/api/v1/users", CreateUser),
            ahserver.NewRoute("GET", "/api/v1/users/:id", GetUser),
        }
    }),
)

    注册全局中间件¶

    // 注册全局中间件（所有路由生效）
fx.Provide(
    ahserver.NewFxServerMiddlewareProvider(func() app.HandlerFunc {
        return func(ctx context.Context, c *app.RequestContext) {
            start := time.Now()
            c.Next(ctx)
            slog.Info("request", "path", string(c.Path()), "elapsed", time.Since(start))
        }
    }),
)

    注册增强能力¶

    fx.Provide(
    ahserver.NewFxServerOptionsProvider(func() *ahserver.ServerOption {
        return ahserver.WithPprof("/debug/pprof")
    }),
)

    客户端开发¶

    使用 HertzClient 直接调用¶

    type MyService struct {
    hertzClient *client.Client
}

// 注入 Hertz Client（由 FX 自动提供）
func NewMyService(cli *client.Client) *MyService {
    return &MyService{hertzClient: cli}
}

// 发起 GET 请求
func (s *MyService) GetUser(ctx context.Context, id string) (*User, error) {
    var user User
    _, body, err := s.hertzClient.Get(ctx, nil, "http://my-service/api/v1/users/" + id)
    if err != nil {
        return nil, err
    }
    json.Unmarshal(body, &user)
    return &user, nil
}

    使用 HertzBaseClient（推荐）¶

    import ahclient "github.com/aif-go/ag-core/contribute/aghertz/aghertzclient"

type MyService struct {
    baseClient *ahclient.HertzBaseClient
}

func NewMyService(cli *client.Client) *MyService {
    return &MyService{
        baseClient: ahclient.NewHertzBaseClient(
            cli,
            ahclient.WithSDEndpoint("my-service"),  // 服务发现模式
        ),
    }
}

func (s *MyService) GetUser(ctx context.Context, id string) (*User, error) {
    var resp GetUserResponse
    err := s.baseClient.DoRequest(ctx, &ahclient.RequestParam{
        Method: "GET",
        Path:   "/api/v1/users/:id",
        PathVars: map[string]string{"id": id},
    }, nil, &resp)
    return &resp.User, err
}

    带请求体的 POST 请求¶

    func (s *MyService) CreateUser(ctx context.Context, req *CreateUserRequest) (*User, error) {
    var resp CreateUserResponse
    err := s.baseClient.DoRequest(ctx, &ahclient.RequestParam{
        Method:   "POST",
        Path:     "/api/v1/users",
    }, req, &resp)  // req 自动序列化为 JSON
    return &resp.User, err
}

    服务发现¶

    启用 Nacos¶

    确保 Nacos 模块已注册，并在配置中启用：

    # Nacos client 配置（由 agnacos 模块提供）
nacos:
  client:
    serverAddr: 127.0.0.1:8848

# Hertz 注册
hertz:
  server:
    serviceName: my-service
    registry:
      cluster: DEFAULT
      group: DEFAULT_GROUP

# Hertz 发现
  client:
    discovery:
      enabled: true
      type: nacos

    最佳实践¶

    ✅ 用 HertzBaseClient 封装服务调用¶

    使用 HertzBaseClient 而非直接操作 *client.Client，可以获得： - 路径参数替换（:id → 实际值） - 自动序列化/反序列化 - 智能路径拼接（防双斜杠） - 统一错误处理

    ✅ 服务发现 vs 直连¶

    // 服务发现（推荐，生产环境）
WithSDEndpoint("service-name")

// 直连（开发/测试环境）
WithDirectEndpoint("http://127.0.0.1:7000")

    ✅ 中间件使用优先级¶

    推荐的中间件优先级： 

    const (
    // 日志/监控（最先执行，捕获原始请求）
    MyLogging    = ClientMiddlewarePriorityHighest       // 0

    // 鉴权/认证
    MyAuth       = ClientMiddlewarePriorityHigh          // 1000

    // 元数据透传、服务发现
    MyMetadata   = ClientMiddlewarePriorityNormal        // 2000

    // 重试/熔断（最后执行）
    MyRetry      = ClientMiddlewarePriorityLow           // 3000
)

    ✅ 用 group 标签注册路由¶

    不要在 ServerConfigurator 中手动添加路由，而是通过 FX group 注入——这样路由注册与业务代码解耦，也便于测试。

    ❌ 不要直接操作原生 Hertz 配置¶

    // ❌ 不要手动创建 *server.Hertz
h := server.New(server.WithHostPorts(":7000"))
h.Spin()

// ✅ 通过 FX 模块自动装配
ahserver.FxAgHertzServerModule

    FAQ¶

    Q: 如何自定义序列化器？ A: 实现 Serializer 接口后调用 RegisterSerializer： 

    ahclient.RegisterSerializer("application/xml", &MyXmlSerializer{})

    Q: 如何关闭服务发现，使用直连？ A: Client 配置 discovery.enabled: false，或在 BaseClient 中使用 WithDirectEndpoint("http://host:port")。

    Q: 如何配置 H2C（HTTP/2 Cleartext）？ A: Server 配置 enableH2C: true，它自动注册 WithH2C() 增强选项。

    Q: 如何集成自定义注册中心？ A: 实现 registry.Registry 接口，FX 注入到 FxAgHertzServerModule 的 Register 参数（通过 HertzServerConfigParam）。
